// ==========================================
// 2b. NAVEGACIÓN CON URLs LIMPIAS (History API)
// ==========================================
// Los enlaces siguen usando href="#calc-id" para que la navegación CSS-first
// (.view:target) funcione incluso SIN JavaScript. Con JS activo, interceptamos
// el clic y sustituimos la URL por una ruta real (/calculadora-id/), que es la
// misma ruta que genera el sitemap. Así sitemap y navegación quedan alineados.
function slugToPath(dataNav) {
  if (dataNav === 'home') return '/';
  const slug = dataNav.replace('calc-', '');
  return `/calculadora-${slug}/`;
}

function initPrettyUrlNav() {
  document.addEventListener('click', (e) => {
    const link = e.target.closest('[data-nav]');
    if (!link) return;
    e.preventDefault();
    const nav = link.getAttribute('data-nav');
    const path = slugToPath(nav);
    if (window.location.pathname !== path) {
      history.pushState({ nav }, '', path);
    }
    applyView(nav === 'home' ? 'home' : nav);
  });

  window.addEventListener('popstate', () => {
    resolveCurrentView();
  });
}

// Aplica una vista concreta (por id de sección) y actualiza SEO/scroll
function applyView(targetId) {
  document.querySelectorAll('.view').forEach(el => el.style.display = 'none');
  const el = document.getElementById(targetId);
  if (el) {
    el.style.display = 'block';
    window.scrollTo({ top: 0, behavior: 'smooth' });
  } else {
    document.getElementById('home').style.display = 'block';
  }
  if (targetId.startsWith('calc-')) {
    updateSEO(targetId.replace('calc-', ''));
  } else {
    updateSchemaHome();
    document.getElementById('meta-canonical').href = window.location.origin + '/';
  }
}

// Resuelve qué vista mostrar según la ruta actual (usado en popstate y carga inicial)
function resolveCurrentView() {
  const path = window.location.pathname;
  const match = path.match(/calculadora-([\w-]+)\/?$/);
  if (match && toolsData[match[1]]) {
    applyView('calc-' + match[1]);
  } else {
    applyView('home');
  }
}

function initRouter() {
  const path = window.location.pathname;
  const hash = window.location.hash;
  let targetId = 'home';

  // 1. Soporte para URL limpia (ej: /calculadora-hipoteca/) si el servidor está configurado
  if (path.includes('calculadora-')) {
    const slug = path.split('calculadora-')[1].replace('/', '').replace('.html', '');
    if (toolsData[slug]) {
      targetId = 'calc-' + slug;
      updateSEO(slug);
    }
  } 
  // 2. Soporte nativo para hash (ej: #calc-hipoteca) - Funciona sin servidor
  else if (hash && hash.startsWith('#calc-')) {
    const slug = hash.replace('#calc-', '');
    if (toolsData[slug]) {
      targetId = hash.substring(1);
      updateSEO(slug);
    }
  }

  // Aplicar vista (CSS :target se encarga de la mayoría, esto es para el estado inicial)
  if (targetId !== 'home') {
    document.querySelectorAll('.view').forEach(el => el.style.display = 'none');
    const targetEl = document.getElementById(targetId);
    if (targetEl) {
      targetEl.style.display = 'block';
      window.scrollTo({ top: 0, behavior: 'smooth' });
    }
  }
}

// 4. PARSER SEGURO (safeEval) - SIN eval() NI Function()
// ==========================================
function safeEval(expr) {
  const s = String(expr).replace(/\s+/g, '');
  let pos = 0;
  const peek = () => s[pos];
  const fail = m => { throw new Error(m); };
  function factorial(n) {
    if (n < 0 || !Number.isInteger(n)) fail('Factorial solo enteros ≥ 0');
    if (n > 170) fail('Número demasiado grande');
    let r = 1; for (let i = 2; i <= n; i++) r *= i; return r;
  }
  function parseExpr() {
    let v = parseTerm();
    while (peek() === '+' || peek() === '-') { const o = s[pos++]; v = o === '+' ? v + parseTerm() : v - parseTerm(); }
    return v;
  }
  function parseTerm() {
    let v = parseFactor();
    while (true) {
      if (peek() === '*' && s[pos + 1] !== '*') { pos++; v *= parseFactor(); }
      else if (peek() === '/') { pos++; v /= parseFactor(); }
      else break;
    }
    return v;
  }
  function parseFactor() {
    if (peek() === '-') { pos++; return -parseFactor(); }
    if (peek() === '+') { pos++; return parseFactor(); }
    let v = parseBase();
    while (peek() === '!' || peek() === '%') {
      if (peek() === '!') { pos++; v = factorial(v); } else { pos++; v = v / 100; }
    }
    if ((s[pos] === '*' && s[pos + 1] === '*') || peek() === '^') {
      pos += peek() === '^' ? 1 : 2;
      v = Math.pow(v, parseFactor());
    }
    return v;
  }
  function parseBase() {
    if (peek() === '(') { pos++; const v = parseExpr(); if (peek() !== ')') fail('Falta )'); pos++; return v; }
    const fns = { sin: Math.sin, cos: Math.cos, tan: Math.tan, sqrt: Math.sqrt, log: Math.log10, ln: Math.log };
    for (const n of Object.keys(fns)) {
      if (s.startsWith(n, pos) && s[pos + n.length] === '(') {
        pos += n.length + 1; const a = parseExpr();
        if (peek() !== ')') fail('Falta )'); pos++;
        return fns[n](a);
      }
    }
    if (s.startsWith('PI', pos)) { pos += 2; return Math.PI; }
    const m = /^\d+(\.\d+)?/.exec(s.slice(pos));
    if (m) { pos += m[0].length; return parseFloat(m[0]); }
    fail('Carácter inesperado');
  }
  const r = parseExpr();
  if (pos !== s.length) fail('Expresión inválida');
  return r;
}

function gcd(a, b) { a = Math.abs(a); b = Math.abs(b); while (b) { [a, b] = [b, a % b]; } return a; }

function calcFracciones(e) {
  e.preventDefault();
  const n1 = val('fr-n1'), d1 = val('fr-d1'), n2 = val('fr-n2'), d2 = val('fr-d2'), op = document.getElementById('fr-op').value;
  if ([n1,d1,n2,d2].some(v => v === null) || d1 === 0 || d2 === 0) { setText('res-fracciones', 'Denominador no puede ser 0'); return; }
  let rn, rd;
  if (op === '+') { rn = n1*d2 + n2*d1; rd = d1*d2; }
  else if (op === '-') { rn = n1*d2 - n2*d1; rd = d1*d2; }
  else if (op === '*') { rn = n1*n2; rd = d1*d2; }
  else { if (n2 === 0) { setText('res-fracciones', 'No se puede dividir por una fracción con numerador 0'); return; } rn = n1*d2; rd = d1*n2; }
  const g = gcd(rn, rd) || 1;
  rn /= g; rd /= g;
  if (rd < 0) { rn = -rn; rd = -rd; }
  setText('res-fracciones', `Resultado: ${rn}/${rd} (≈ ${(rn/rd).toFixed(4)})`);
}

  const sorted = [...nums].sort((a,b)=>a-b);

  nums.forEach(n => freq[n] = (freq[n]||0)+1);
  const maxFreq = Math.max(...Object.values(freq));
  const moda = maxFreq > 1 ? Object.keys(freq).filter(k => freq[k]===maxFreq).join(', ') : 'Sin moda';

  setText('res-pr-mediana', mediana.toFixed(4));

function calcHipoteca(e) {
  e.preventDefault();
  const precio = val('hip-val'), entrada = val('hip-ini'), anos = val('hip-anos'), tasa = val('hip-tasa');
  if (!precio || !entrada || !anos || tasa === null || precio <= entrada) return;
  const principal = precio - entrada, n = Math.round(anos * 12), r = tasa / 100 / 12;
  const cuota = r === 0 ? principal / n : principal * r * Math.pow(1 + r, n) / (Math.pow(1 + r, n) - 1);
  document.getElementById('res-hip-cuota').textContent = fmt(cuota);
  document.getElementById('res-hip-int').textContent = fmt((cuota * n) - principal);
  document.getElementById('res-hip-total').textContent = fmt(cuota * n);

  let balance = principal, tbody = '', yInt = 0, yCap = 0;
  for (let m = 1; m <= n; m++) {
    const interest = balance * r, principalPaid = cuota - interest;
    balance = Math.max(0, balance - principalPaid);
    yInt += interest; yCap += principalPaid;
    if (m % 12 === 0 || m === n) {
      tbody += `<tr><td>Año ${Math.ceil(m/12)}</td><td>${fmt(yInt)}</td><td>${fmt(yCap)}</td><td>${fmt(balance)}</td></tr>`;
      yInt = 0; yCap = 0;
    }
  }
  document.querySelector('#hipoteca-table tbody').innerHTML = tbody;
  document.getElementById('hipoteca-table-container').classList.remove('hidden');
}

      .replace(/,/g, '.')
      .replace(/\s+/g, '');

    if (!s || s.length > 500) throw new Error('Expresión no válida');

    let pos = 0;

    const fail = message => {

    };

    const peek = () => s[pos] ?? '';

    const consume = ch => {
      if (peek() === ch) {
        pos++;

    safePrevent(e);

      out('res-pr-media', '-');
      out('res-pr-mediana', '-');
      out('res-pr-moda', 'Introduce números válidos');
      return;
    }

    const mean = nums.reduce((sum, value) => sum + value, 0) / nums.length;

    const sorted = [...nums].sort((a, b) => a - b);
    const middle = Math.floor(sorted.length / 2);

    const median = sorted.length % 2
      ? sorted[middle]
      : (sorted[middle - 1] + sorted[middle]) / 2;

    const frequencies = new Map();

    nums.forEach(value => {
      frequencies.set(value, (frequencies.get(value) || 0) + 1);
    });

    const maxFrequency = Math.max(...frequencies.values());

    const modes = maxFrequency > 1
      ? [...frequencies.entries()]
          .filter(([, frequency]) => frequency === maxFrequency)
          .map(([value]) => fmtNum(value))
      : [];

    out('res-pr-media', fmtNum(mean, 8));
    out('res-pr-mediana', fmtNum(median, 8));
    out('res-pr-moda', modes.length ? modes.join(', ') : 'Sin moda');

    const principal = price - downPayment;
    const months = Math.max(1, Math.round(years * 12));
    const r = annualRate / 100 / 12;

    const payment = r === 0
      ? principal / months
      : principal * r * Math.pow(1 + r, months) /
        (Math.pow(1 + r, months) - 1);

    const total = payment * months;
    const interestTotal = total - principal;

    out('res-hip-cuota', fmtMoney(payment));
    out('res-hip-int', fmtMoney(interestTotal));

      yearCapital += capitalPaid;

            <td>${fmtMoney(yearInterest)}</td>

    if (!value) return null;

    const [year, month, day] = value.split('-').map(Number);

    if (![year, month, day].every(Number.isInteger)) return null;

    const date = new Date(year, month - 1, day);

    if (
      date.getFullYear() !== year ||
      date.getMonth() !== month - 1 ||
      date.getDate() !== day
    ) {
      return null;

    return date;
  };

  }

  if (typeof toolsData !== 'undefined' && !toolsData.energiaCinetica) {
    toolsData.energiaCinetica = {
      cat: 'ciencia',
      icon: '⚡',
      label: 'Energía Cinética',
      title: 'Calculadora de Energía Cinética',
      desc: 'Calcula la energía cinética a partir de masa y velocidad.',
      seo: 'Calcula la energía cinética mediante la fórmula Ec = ½mv².',
      example: '2 kg a 3 m/s = 9 J',
      table: null,
      faqs: [
        {
          q: '¿Qué fórmula utiliza?',
          a: 'La fórmula Ec = ½ × masa × velocidad².'
        }
      ]
    };
  }

  /* ---------- Router robusto ---------- */
  function getPathCalculator() {
    const path = window.location.pathname.replace(/\/+$/, '');

    const match = path.match(/^\/calculadora-([^/]+)$/);

    if (!match) return null;

    const id = decodeURIComponent(match[1]);

    return typeof toolsData !== 'undefined' && toolsData[id]
      ? id
      : null;
  }

  window.slugToPath = function (nav) {
    if (nav === 'home') return '/';

    const id = String(nav).replace(/^calc-/, '');

    return `/calculadora-${id}/`;
  };

  window.applyView = function (targetId) {
    document.querySelectorAll('.view').forEach(section => {
      section.style.display = 'none';
    });

    const home = $('home');

    if (targetId === 'home') {
      if (home) home.style.display = 'block';

      if (typeof window.updateSchemaHome === 'function') {
        window.updateSchemaHome();
      }

      const canonical = $('meta-canonical');

      if (canonical) {
        canonical.href = `${window.location.origin}/`;
      }

      return;
    }

    const calculator = $(targetId);

    if (!calculator) {
      if (home) home.style.display = 'block';
      return;
    }

    calculator.style.display = 'block';

    const id = targetId.replace(/^calc-/, '');

    if (typeof window.updateSEO === 'function' && typeof toolsData !== 'undefined' && toolsData[id]) {
      window.updateSEO(id);
    }

    window.scrollTo({ top: 0, behavior: 'smooth' });
  };

  window.resolveCurrentView = function () {
    const id = getPathCalculator();

    if (id) {
      window.applyView(`calc-${id}`);
      return;
    }

    const hash = window.location.hash;

    if (hash.startsWith('#calc-')) {
      const hashId = hash.slice(6);

      if (typeof toolsData !== 'undefined' && toolsData[hashId]) {
        window.applyView(`calc-${hashId}`);
        return;
      }
    }

    window.applyView('home');
  };

  window.initRouter = function () {
    window.resolveCurrentView();
  };

  window.initPrettyUrlNav = function () {
    if (window.__multiCalcRouterInstalled) return;
    window.__multiCalcRouterInstalled = true;

    document.addEventListener('click', event => {
      const link = event.target.closest('[data-nav]');

      if (!link) return;

      const nav = link.getAttribute('data-nav');

      if (!nav) return;

      event.preventDefault();

      const path = window.slugToPath(nav);

      if (window.location.pathname !== path) {
        history.pushState(
          { nav },
          '',
          path
        );
      }

      window.applyView(
        nav === 'home' ? 'home' : `calc-${String(nav).replace(/^calc-/, '')}`
      );
    });

    window.addEventListener('popstate', window.resolveCurrentView);

    window.addEventListener('hashchange', () => {
      if (window.location.hash.startsWith('#calc-')) {
        window.resolveCurrentView();
      }
    });
  };

  /* ---------- Inicialización ---------- */
  function repairCalculatorState() {
    window.toggleAreaFields?.();
    window.togglePotLabel?.();
    window.toggleOhmFields?.();

    const home = $('home');

    if (getPathCalculator()) {
      window.resolveCurrentView();
    } else if (window.location.hash.startsWith('#calc-')) {
      window.resolveCurrentView();
    } else if (home) {
      window.applyView('home');
    }

    /*
     * Las tasas se cargan después de que el DOM esté listo.
     */
    updateCurrencyRates();
  }

  /*
   * El script original registra DOMContentLoaded para initApp.
   * Este bloque se ejecuta después de ese registro, pero antes
   * del evento, por lo que las funciones sobrescritas son las
   * que usará initApp cuando llegue DOMContentLoaded.
   */
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', repairCalculatorState, {
      once: true
    });
  } else {
    repairCalculatorState();
  }

})();
</script>
